# fnos 0.4.0 Complete Command Upgrade Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Upgrade fnos-cli to `fnos ^0.4.0`, expose all 111 SDK business methods, support two-factor/token/WSS authentication, and verify the real CLI against a pinned fnos-mock-server.

**Architecture:** Replace the positional `COMMAND_MAPPING` array with six explicit domain catalogs and a shared option compiler. Keep connection/authentication in `src/utils/client.js`, isolate command execution in `src/commands/runner.js`, and make the CLI entry point import-safe. Use offline unit/contract tests for deterministic behavior and subprocess smoke tests for the complete WebSocket path.

**Tech Stack:** Node.js 20, CommonJS, Commander 11, fnos 0.4.0, Winston 3, Mocha 10, Python 3.11, uv, fnos-mock-server commit `d9592a05a8e07082b954921acfae3a9a915f3c01`.

## Global Constraints

- Work only in branch `feature/fnos-0.4.0-commands` and its isolated worktree.
- Preserve all 26 existing command names and existing camelCase option spellings.
- The final catalog must contain exactly 21 domains and 111 unique business commands.
- New option spellings use kebab-case; booleans accept only `true` or `false`.
- Business argument validation happens before any network connection.
- Passwords, two-factor codes, tokens, long tokens, access tokens, secrets, and HMAC material must never be logged.
- CLI exit codes remain `0` success, `1` general/connection error, `3` authentication error, and `4` business parameter error.
- `package.json` and `package-lock.json` must both report fnos-cli `0.3.0` at release-documentation completion.
- Commit messages use English Conventional Commits and must not contain `Co-Authored-By`.
- Smoke tests use only `127.0.0.1:5666`, `admin/admin`, `twofauser`, and the pinned mock-server commit; never use a real NAS.

const { describe, it } = require('mocha');

const LEGACY_COMMANDS = [
  'resmon.cpu', 'resmon.gpu', 'resmon.mem', 'resmon.disk', 'resmon.net', 'resmon.gen',
  'store.general', 'store.calcSpace', 'store.listDisk', 'store.diskSmart', 'store.state',

  'file.ls', 'file.mkdir', 'file.rm', 'sac.upsStatus'

describe('complete command catalog', () => {
  it('contains 21 domains and 111 unique commands', () => {
    const names = Object.entries(COMMAND_CATALOG).flatMap(([domainName, domain]) =>
      Object.keys(domain.commands).map((commandName) => `${domainName}.${commandName}`)
    );
    assert.equal(DOMAIN_COUNT, 21);
    assert.equal(COMMAND_COUNT, 111);
    assert.equal(new Set(names).size, 111);
    for (const legacyName of LEGACY_COMMANDS) assert.ok(names.includes(legacyName), legacyName);
  });

  it('references classes and methods exported by fnos 0.4.0', () => {
    assert.equal(sdk.version, '0.4.0');
    for (const [domainName, domain] of Object.entries(COMMAND_CATALOG)) {
      const Constructor = sdk[domain.className];
      assert.equal(typeof Constructor, 'function', `${domainName}: ${domain.className}`);
      for (const [commandName, definition] of Object.entries(domain.commands)) {
        assert.equal(typeof Constructor.prototype[definition.method], 'function', `${domainName}.${commandName}`);
        assert.equal(typeof definition.description, 'string');
        assert.ok(definition.description.length > 0);
        assert.ok(['first', 'last'].includes(definition.timeoutPosition));
        assert.ok(Array.isArray(definition.options));
      }
    }
  });
});

```js
const { Option } = require('commander');

class CommandParameterError extends Error {
  constructor(message) {
    super(message);
    this.name = 'CommandParameterError';
    this.exitCode = 4;
  }
}

function cloneDefault(value) {
  if (Array.isArray(value)) return value.map((item) => cloneDefault(item));
  if (value && typeof value === 'object') return { ...value };
  return value;
}

function parseBoolean(name, value) {
  if (typeof value === 'boolean') return value;
  if (typeof value === 'string' && value.toLowerCase() === 'true') return true;
  if (typeof value === 'string' && value.toLowerCase() === 'false') return false;
  throw new CommandParameterError(`--${name} must be true or false.`);
}

function parseInteger(definition, value) {
  if (typeof value === 'string' && !/^-?\d+$/.test(value.trim())) {
    throw new CommandParameterError(`--${definition.name} must be a decimal integer.`);
  }
  const parsed = typeof value === 'number' ? value : Number(value);
  if (!Number.isInteger(parsed)) throw new CommandParameterError(`--${definition.name} must be an integer.`);
  if (definition.min !== undefined && parsed < definition.min) {
    throw new CommandParameterError(`--${definition.name} must be at least ${definition.min}.`);
  }
  if (definition.max !== undefined && parsed > definition.max) {
    throw new CommandParameterError(`--${definition.name} must be at most ${definition.max}.`);
  }
  if (definition.allowedValues && !definition.allowedValues.includes(parsed)) {
    throw new CommandParameterError(`--${definition.name} must be one of: ${definition.allowedValues.join(', ')}.`);
  }
  return parsed;
}

function parseJson(name, value) {
  if (value && typeof value === 'object') return value;
  try {
    return JSON.parse(value);
  } catch {
    throw new CommandParameterError(`--${name} must be valid JSON.`);
  }
}

function parseValue(definition, value) {
  if (definition.type === 'boolean') return parseBoolean(definition.name, value);
  if (definition.type === 'integer') return parseInteger(definition, value);
  if (definition.type === 'csv') {
    const values = Array.isArray(value) ? value : String(value).split(',').map((item) => item.trim()).filter(Boolean);
    if (values.length === 0) throw new CommandParameterError(`--${definition.name} must contain at least one value.`);
    return values;
  }
  if (definition.type === 'json') {
    const parsed = parseJson(definition.name, value);
    if (!parsed || Array.isArray(parsed) || typeof parsed !== 'object') {
      throw new CommandParameterError(`--${definition.name} must be a JSON object.`);
    }
    return parsed;
  }
  if (definition.type === 'json-record-array') {
    const parsed = parseJson(definition.name, value);
    if (!Array.isArray(parsed) || parsed.length === 0 || parsed.some((item) => !item || Array.isArray(item) || typeof item !== 'object')) {
      throw new CommandParameterError(`--${definition.name} must be a non-empty JSON object array.`);
    }
    return parsed;
  }
  const parsed = String(value);
  if (!parsed.trim()) throw new CommandParameterError(`--${definition.name} must not be empty.`);
  return parsed;
}

function optionReadNames(definition) {
  if (definition.readNames) return definition.readNames;
  return [new Option(definition.flag).attributeName()];
}

function readRawValue(definition, options) {
  for (const name of optionReadNames(definition)) {
    if (options[name] !== undefined) return options[name];
  }
  return undefined;
}

function buildMethodArguments(definition, options, timeout = 10000) {
  const values = definition.options.map((option) => {
    const raw = readRawValue(option, options);
    if (raw === undefined) {
      if (option.required) throw new CommandParameterError(`--${option.name} is required.`);
      return cloneDefault(option.defaultValue);
    }
    return parseValue(option, raw);
  });
  return definition.timeoutPosition === 'first' ? [timeout, ...values] : [...values, timeout];
}

function registerBusinessOptions(command, definitions) {
  for (const definition of definitions) {
    command.addOption(new Option(definition.flag, definition.description));
    for (const legacyFlag of definition.legacyFlags || []) {
      command.addOption(new Option(legacyFlag, `${definition.description} (legacy spelling)`).hideHelp());
    }
  }
  return command;
}

function registerAuthenticationOptions(command, requiredCredentials = false) {
  const add = requiredCredentials ? 'requiredOption' : 'option';
  command[add]('-e, --endpoint <endpoint>', 'Server endpoint, including optional ws:// or wss:// prefix');
  command[add]('-u, --username <username>', 'Username');
  command[add]('-p, --password <password>', 'Password');
  command.option('--twofa-code <code>', 'Six-digit two-factor code');
  command.option('--trust-device', 'Trust this device after two-factor login');
  command.option('--verify-ssl', 'Verify the WSS server certificate');
  return command;
}

module.exports = {
  CommandParameterError,
  buildMethodArguments,
  registerAuthenticationOptions,
  registerBusinessOptions
};
```

const fs = require('node:fs');

  constructor(homeDir = os.homedir()) {
    this.settingsDir = path.join(homeDir, '.fnos');
    this.settingsPath = path.join(this.settingsDir, 'settings.json');
  }

  _ensureDir() {
    fs.mkdirSync(this.settingsDir, { recursive: true, mode: 0o700 });
  }

  load() {
    if (!fs.existsSync(this.settingsPath)) return null;
    try {
      return JSON.parse(fs.readFileSync(this.settingsPath, 'utf8'));
    } catch (error) {
      console.error('Failed to load settings:', error.message);
      return null;
    }
  }

  save(data) {
    this._ensureDir();
    try {
      fs.writeFileSync(this.settingsPath, JSON.stringify(data, null, 2), { mode: 0o600 });
      fs.chmodSync(this.settingsPath, 0o600);
    } catch (error) {
      console.error('Failed to save settings:', error.message);
      throw error;
    }
  }

  clear() {
    if (fs.existsSync(this.settingsPath)) fs.unlinkSync(this.settingsPath);
  }

  clearCredentials() {
    const existingSettings = this.load();
    if (!existingSettings) return;
    const authFields = [
      'endpoint', 'username', 'password', 'token', 'longToken', 'secret',
      'credentialVersion', 'secretFormat'
    ];
    authFields.forEach((field) => delete existingSettings[field]);
    this.save(existingSettings);
  }

  exists() {
    return fs.existsSync(this.settingsPath);
  }

  getCredentials() {
    const data = this.load();
    if (!data) return null;
    const fields = [
      'endpoint', 'username', 'password', 'token', 'longToken', 'secret',

const sdk = require('fnos');
const settings = require('./settings');
const { logger } = require('./logger');

const SDK_CLASS_NAMES = [
  'ResourceMonitor', 'Store', 'SystemInfo', 'User', 'Network', 'File', 'SAC',
  'DockerManager', 'EventLogger', 'Share', 'Notify', 'IscsiManager',
  'BackupManager', 'DownloadCenter', 'IPBlocker', 'LicenseManager',
  'LiveUpdate', 'MountManager', 'NetworkServer', 'Security', 'SystemRestore'
];

class AuthenticationError extends Error {
  constructor(message) {
    super(message);
    this.name = 'AuthenticationError';
    this.exitCode = 3;
  }
}

function canUseToken(credentials) {
  return credentials.credentialVersion === 2 &&
    credentials.secretFormat === 'decrypted-v1' &&
    [credentials.token, credentials.longToken, credentials.secret]
      .every((value) => typeof value === 'string' && value.length > 0);
}

function tokenLoginSucceeded(result) {
  return Boolean(result && (result.result === 'succ' || result.errno === 0));
}

async function passwordLogin(client, credentials, options) {
  if (!credentials.password) throw new AuthenticationError('Password is required.');
  let result = await client.login(credentials.username, credentials.password, 10000);
  if (result.twofaSetupRequired) {
    throw new AuthenticationError('This account must bind two-factor authentication in the fnOS web interface first.');
  }
  if (result.twofaRequired) {
    if (!/^\d{6}$/.test(options.twofaCode || '')) {
      throw new AuthenticationError('A six-digit --twofa-code is required for this account.');
    }
    result = await client.submitTwofaCode(options.twofaCode, options.trustDevice === true, 10000);
  }
  if (!result || result.result !== 'succ' || !result.token || !result.secret) {
    throw new AuthenticationError(result?.msg || result?.errmsg || 'Login failed.');
  }
  return result;
}

async function createClientWithCredentials(credentials, options = {}, dependencies = {}) {
  const sdkModule = dependencies.sdk || sdk;
  const settingsStore = dependencies.settings || settings;
  const activeLogger = dependencies.logger || logger;
  const Client = sdkModule.FnosClient;
  const client = new Client();
  const timeout = options.timeout ?? 60;
  try {
    activeLogger.info(`Connecting to ${credentials.endpoint}...`);
    await client.connect(credentials.endpoint, timeout * 1000, false, options.verifySsl === true ? false : true);
    let result;
    let usedToken = false;
    if (canUseToken(credentials)) {
      result = await client.loginViaToken(credentials.token, credentials.longToken, credentials.secret, 10000);
      usedToken = tokenLoginSucceeded(result);
      if (!usedToken) activeLogger.debug('Saved token was rejected; falling back to password login.');
    }
    if (!usedToken) {
      result = await passwordLogin(client, credentials, options);
      if (options.saveCredentials || credentials.source === 'settings_file') {
        settingsStore.saveCredentials({
          endpoint: credentials.endpoint,
          username: credentials.username,
          password: credentials.password,
          credentialVersion: 2,
          token: result.token || null,
          longToken: result.longToken || null,
          secret: client.getDecryptedSecret(),
          secretFormat: 'decrypted-v1'
        });
      }
    }
    return client;
  } catch (error) {
    client.close();
    throw error;
  }
}

async function createClient(options = {}, dependencies = {}) {
  const credentials = options.credentials || {
    endpoint: options.endpoint,
    username: options.username,
    password: options.password,
    source: options.saveCredentials ? 'settings_file' : 'command_line'
  };
  if (!credentials.endpoint || !credentials.username) {
    throw new AuthenticationError('Missing credentials. Run "fnos login" or provide -e/-u/-p.');
  }
  return createClientWithCredentials(credentials, options, dependencies);
}

function getSDKInstance(client, className, sdkModule = sdk) {
  if (!SDK_CLASS_NAMES.includes(className) || typeof sdkModule[className] !== 'function') {
    throw new Error(`Unknown SDK class: ${className}`);
  }
  return new sdkModule[className](client);
}

module.exports = {
  AuthenticationError,
  SDK_CLASS_NAMES,
  createClient,
  createClientWithCredentials,
  getSDKInstance
};

- [ ] **Step 3: Implement deterministic command-reference rendering**

```js
const fs = require('node:fs');
const path = require('node:path');
const { COMMAND_CATALOG, COMMAND_COUNT, DOMAIN_COUNT } = require('../src/commands/catalog');

function displayDefault(value) {
  if (value === undefined) return '';
  return ` Default: \`${JSON.stringify(value)}\`.`;
}

function renderCommandReference() {
  const lines = [
    '# Command Reference',
    '',
    `fnos-cli exposes ${COMMAND_COUNT} commands across ${DOMAIN_COUNT} domains.`,
    ''
  ];
  for (const [domainName, domain] of Object.entries(COMMAND_CATALOG)) {
    lines.push(`## ${domainName}`, '');
    for (const [commandName, definition] of Object.entries(domain.commands)) {
      lines.push(`### \`${domainName}.${commandName}\``, '', definition.description, '');
      if (definition.options.length === 0) {
        lines.push('No business options.', '');
      } else {
        lines.push('| Option | Required | Type | Description |', '|---|---:|---|---|');
        for (const option of definition.options) {
          lines.push(`| \`${option.flag}\` | ${option.required ? 'yes' : 'no'} | ${option.type} | ${option.description}.${displayDefault(option.defaultValue)} |`);
        }
        lines.push('');
      }
    }
  }
  return `${lines.join('\n')}\n`;
}

if (require.main === module) {
  fs.writeFileSync(path.resolve(__dirname, '../docs/commands.md'), renderCommandReference());
}

module.exports = { renderCommandReference };
```
